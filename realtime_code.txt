import json
from pathlib import Path
from collections import deque, Counter

import cv2
import numpy as np
import mediapipe as mp
from tensorflow.keras.models import load_model


# ============================================================
# CONFIGURATION
# ============================================================

BASE_DIR = Path(__file__).resolve().parents[2]

MODEL_PATH = BASE_DIR / "models" / "include50_lstm_v2_best.keras"
CLASSES_PATH = BASE_DIR / "models" / "include50_classes_v2.json"
SIGN_INFO_PATH = BASE_DIR / "app" / "data" / "sign_info.json"

SEQUENCE_LENGTH = 45
FEATURES = 258

CONFIDENCE_THRESHOLD = 0.50
SMOOTHING_WINDOW = 7


# ============================================================
# COLORS
# ============================================================

WHITE = (255, 255, 255)
BLACK = (0, 0, 0)
GREEN = (0, 255, 0)
RED = (0, 0, 255)
YELLOW = (0, 255, 255)
BLUE = (255, 150, 0)


# ============================================================
# MEDIAPIPE
# ============================================================

mp_holistic = mp.solutions.holistic
mp_drawing = mp.solutions.drawing_utils


# ============================================================
# LOAD JSON
# ============================================================

def load_classes():
    with open(CLASSES_PATH, "r", encoding="utf-8") as f:
        data = json.load(f)

    if isinstance(data, list):
        return data

    if isinstance(data, dict):
        # Handle common formats
        for key in ["classes", "class_names", "labels"]:
            if key in data:
                value = data[key]
                if isinstance(value, list):
                    return value

        # If dictionary is index -> class
        try:
            return [
                value
                for key, value in sorted(
                    data.items(),
                    key=lambda x: int(x[0])
                )
            ]
        except Exception:
            return list(data.values())

    raise ValueError("Invalid classes JSON format.")


def load_sign_info():
    if not SIGN_INFO_PATH.exists():
        return {}

    with open(SIGN_INFO_PATH, "r", encoding="utf-8") as f:
        return json.load(f)


# ============================================================
# MEDIAPIPE LANDMARK EXTRACTION
# ============================================================

def extract_keypoints(results):
    """
    258 features:
    Pose      = 33 * 4  = 132
    Left hand = 21 * 3  = 63
    Right hand= 21 * 3  = 63
    """

    pose = np.array(
        [
            [lm.x, lm.y, lm.z, lm.visibility]
            for lm in results.pose_landmarks.landmark
        ],
        dtype=np.float32
    ).flatten() if results.pose_landmarks else np.zeros(132, dtype=np.float32)

    left_hand = np.array(
        [
            [lm.x, lm.y, lm.z]
            for lm in results.left_hand_landmarks.landmark
        ],
        dtype=np.float32
    ).flatten() if results.left_hand_landmarks else np.zeros(63, dtype=np.float32)

    right_hand = np.array(
        [
            [lm.x, lm.y, lm.z]
            for lm in results.right_hand_landmarks.landmark
        ],
        dtype=np.float32
    ).flatten() if results.right_hand_landmarks else np.zeros(63, dtype=np.float32)

    keypoints = np.concatenate(
        [pose, left_hand, right_hand]
    )

    if len(keypoints) != FEATURES:
        keypoints = np.resize(keypoints, FEATURES)

    return keypoints


# ============================================================
# DRAW FUNCTIONS
# ============================================================

def draw_text(
    frame,
    text,
    position,
    color=WHITE,
    scale=0.7,
    thickness=2
):
    cv2.putText(
        frame,
        str(text),
        position,
        cv2.FONT_HERSHEY_SIMPLEX,
        scale,
        color,
        thickness,
        cv2.LINE_AA
    )


def draw_landmarks(frame, results):
    if results.pose_landmarks:
        mp_drawing.draw_landmarks(
            frame,
            results.pose_landmarks,
            mp_holistic.POSE_CONNECTIONS
        )

    if results.left_hand_landmarks:
        mp_drawing.draw_landmarks(
            frame,
            results.left_hand_landmarks,
            mp_holistic.HAND_CONNECTIONS
        )

    if results.right_hand_landmarks:
        mp_drawing.draw_landmarks(
            frame,
            results.right_hand_landmarks,
            mp_holistic.HAND_CONNECTIONS
        )


def draw_prediction(
    frame,
    predicted_sign,
    confidence,
    selected_sign
):

    h, w = frame.shape[:2]

    # Prediction box
    cv2.rectangle(
        frame,
        (10, 50),
        (min(w - 10, 630), 145),
        BLACK,
        -1
    )

    draw_text(
        frame,
        "PREDICTION",
        (25, 80),
        YELLOW,
        0.65,
        2
    )

    draw_text(
        frame,
        predicted_sign,
        (25, 115),
        GREEN if confidence >= CONFIDENCE_THRESHOLD else RED,
        0.9,
        2
    )

    draw_text(
        frame,
        f"Confidence: {confidence * 100:.1f}%",
        (25, 140),
        WHITE,
        0.55,
        2
    )

    if selected_sign:
        draw_text(
            frame,
            f"Selected: {selected_sign}",
            (25, h - 25),
            BLUE,
            0.65,
            2
        )


def draw_sign_information(frame, selected_sign, sign_info):

    if not selected_sign:
        return

    info = sign_info.get(selected_sign)

    if not info:
        return

    h, w = frame.shape[:2]

    x1 = max(10, w - 390)
    y1 = 50
    x2 = w - 10
    y2 = 205

    overlay = frame.copy()

    cv2.rectangle(
        overlay,
        (x1, y1),
        (x2, y2),
        BLACK,
        -1
    )

    frame[:] = cv2.addWeighted(
        overlay,
        0.75,
        frame,
        0.25,
        0
    )

    draw_text(
        frame,
        info.get("title", selected_sign),
        (x1 + 15, y1 + 35),
        YELLOW,
        0.7,
        2
    )

    draw_text(
        frame,
        "Category: " + info.get("category", "General"),
        (x1 + 15, y1 + 65),
        WHITE,
        0.5,
        1
    )

    instruction = info.get(
        "instruction",
        "Perform the sign shown in the reference video."
    )

    # Wrap instruction
    words = instruction.split()
    lines = []
    current = ""

    for word in words:
        test = current + " " + word

        if len(test) > 42:
            lines.append(current)
            current = word
        else:
            current = test

    if current:
        lines.append(current)

    y = y1 + 95

    for line in lines[:3]:
        draw_text(
            frame,
            line.strip(),
            (x1 + 15, y),
            WHITE,
            0.45,
            1
        )
        y += 22


def draw_controls(frame):

    h, w = frame.shape[:2]

    text = "Q Quit | R Reset | SPACE Select | H Info"

    draw_text(
        frame,
        text,
        (15, h - 50),
        WHITE,
        0.48,
        1
    )


# ============================================================
# MAIN
# ============================================================

def main():

    print("=" * 70)
    print("INCLUDE-50 V2 REAL-TIME ISL RECOGNITION")
    print("=" * 70)

    # --------------------------------------------------------
    # Load model
    # --------------------------------------------------------

    print("\nLoading V2 model...")

    if not MODEL_PATH.exists():
        print("ERROR: Model not found:")
        print(MODEL_PATH)
        return

    model = load_model(MODEL_PATH)

    print("V2 model loaded successfully.")
    print("Model input :", model.input_shape)
    print("Model output:", model.output_shape)

    # --------------------------------------------------------
    # Load classes
    # --------------------------------------------------------

    if not CLASSES_PATH.exists():
        print("ERROR: Classes file not found:")
        print(CLASSES_PATH)
        return

    class_names = load_classes()

    print("Classes loaded:", len(class_names))

    # --------------------------------------------------------
    # Load sign information
    # --------------------------------------------------------

    sign_info = load_sign_info()

    print("Sign information loaded:", len(sign_info))

    # --------------------------------------------------------
    # Camera
    # --------------------------------------------------------

    print("\nStarting webcam...")

    cap = cv2.VideoCapture(0)

    if not cap.isOpened():
        print("ERROR: Cannot open webcam.")
        return

    cap.set(cv2.CAP_PROP_FRAME_WIDTH, 960)
    cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 720)

    print("\nControls:")
    print("  Q = Quit")
    print("  R = Reset prediction")
    print("  SPACE = Select current prediction")
    print("  H = Show/hide selected sign information")
    print()

    # --------------------------------------------------------
    # State
    # --------------------------------------------------------

    sequence = deque(maxlen=SEQUENCE_LENGTH)
    prediction_history = deque(maxlen=SMOOTHING_WINDOW)

    predicted_sign = "Waiting..."
    predicted_confidence = 0.0

    selected_sign = None
    show_info = True

    # --------------------------------------------------------
    # MediaPipe
    # --------------------------------------------------------

    with mp_holistic.Holistic(
        min_detection_confidence=0.5,
        min_tracking_confidence=0.5,
        model_complexity=1
    ) as holistic:

        while cap.isOpened():

            ret, frame = cap.read()

            if not ret:
                print("Failed to receive webcam frame.")
                break

            # Mirror webcam
            frame = cv2.flip(frame, 1)

            # Convert BGR -> RGB
            image = cv2.cvtColor(
                frame,
                cv2.COLOR_BGR2RGB
            )

            image.flags.writeable = False

            results = holistic.process(image)

            image.flags.writeable = True

            # ------------------------------------------------
            # Draw landmarks
            # ------------------------------------------------

            draw_landmarks(
                frame,
                results
            )

            # ------------------------------------------------
            # Extract 258 features
            # ------------------------------------------------

            keypoints = extract_keypoints(results)

            sequence.append(keypoints)

            # ------------------------------------------------
            # Prediction
            # ------------------------------------------------

            if len(sequence) == SEQUENCE_LENGTH:

                input_data = np.array(
                    sequence,
                    dtype=np.float32
                )

                input_data = np.expand_dims(
                    input_data,
                    axis=0
                )

                try:

                    prediction = model.predict(
                        input_data,
                        verbose=0
                    )[0]

                    index = int(
                        np.argmax(prediction)
                    )

                    confidence = float(
                        prediction[index]
                    )

                    if index < len(class_names):

                        current_sign = class_names[index]

                        prediction_history.append(
                            current_sign
                        )

                        # Majority vote
                        most_common = Counter(
                            prediction_history
                        ).most_common(1)[0][0]

                        predicted_sign = most_common

                        # Use confidence belonging to
                        # current top prediction
                        predicted_confidence = confidence

                except Exception as e:

                    print(
                        "Prediction error:",
                        e
                    )

            # ------------------------------------------------
            # Draw prediction
            # ------------------------------------------------

            draw_prediction(
                frame,
                predicted_sign,
                predicted_confidence,
                selected_sign
            )

            # ------------------------------------------------
            # Sign information
            # ------------------------------------------------

            if show_info and selected_sign:

                draw_sign_information(
                    frame,
                    selected_sign,
                    sign_info
                )

            # ------------------------------------------------
            # Status
            # ------------------------------------------------

            if len(sequence) < SEQUENCE_LENGTH:

                status = (
                    f"Collecting: "
                    f"{len(sequence)}/{SEQUENCE_LENGTH}"
                )

                draw_text(
                    frame,
                    status,
                    (20, 180),
                    YELLOW,
                    0.55,
                    2
                )

            else:

                draw_text(
                    frame,
                    "READY - PERFORM ONE SIGN",
                    (20, 180),
                    GREEN,
                    0.55,
                    2
                )

            # ------------------------------------------------
            # Controls
            # ------------------------------------------------

            draw_controls(frame)

            # ------------------------------------------------
            # SHOW WEBCAM
            # ------------------------------------------------

            cv2.imshow(
                "INCLUDE-50 V2 ISL Recognition",
                frame
            )

            # ------------------------------------------------
            # KEYBOARD
            # ------------------------------------------------

            key = cv2.waitKey(1) & 0xFF

            # Q
            if key == ord("q"):
                break

            # R
            elif key == ord("r"):

                sequence.clear()
                prediction_history.clear()

                predicted_sign = "Waiting..."
                predicted_confidence = 0.0

                print("Prediction reset.")

            # SPACE
            elif key == 32:

                if predicted_sign != "Waiting...":

                    selected_sign = predicted_sign

                    print(
                        f"Selected sign: "
                        f"{selected_sign} "
                        f"({predicted_confidence * 100:.1f}%)"
                    )

            # H
            elif key == ord("h"):

                show_info = not show_info

    # --------------------------------------------------------
    # Cleanup
    # --------------------------------------------------------

    cap.release()

    cv2.destroyAllWindows()

    print("\nWebcam stopped.")
    print("=" * 70)


if __name__ == "__main__":
    main()
